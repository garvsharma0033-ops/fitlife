/* ════════════════════════════════════════════
   USER PROFILE
════════════════════════════════════════════ */
let user = {};
 
function calcStats(u) {
  const h = u.height, w = u.weight, a = u.age, g = u.gender;
  const bmi = w / ((h/100) ** 2);
  // Mifflin-St Jeor BMR
  let bmr = g === 'female'
    ? 10*w + 6.25*h - 5*a - 161
    : 10*w + 6.25*h - 5*a + 5;
  const tdee = Math.round(bmr * u.activity);
  bmr = Math.round(bmr);
  const bmiVal = Math.round(bmi * 10) / 10;
 
  let goalCal = tdee;
  if (u.goal === 'lose') goalCal = Math.max(1200, tdee - 500);
  if (u.goal === 'gain') goalCal = tdee + 300;
  goalCal = Math.round(goalCal);
 
  // Macro split: protein 30%, carbs 40%, fat 30% (lose: 35/35/30, gain: 30/45/25)
  let pSplit = 0.30, cSplit = 0.40, fSplit = 0.30;
  if (u.goal === 'lose')  { pSplit=0.35; cSplit=0.35; fSplit=0.30; }
  if (u.goal === 'gain')  { pSplit=0.30; cSplit=0.45; fSplit=0.25; }
  const protG = Math.round(goalCal * pSplit / 4);
  const carbG = Math.round(goalCal * cSplit / 4);
  const fatG  = Math.round(goalCal * fSplit / 9);
 
  let bmiLabel = 'Healthy weight';
  if (bmiVal < 18.5) bmiLabel = 'Underweight';
  else if (bmiVal >= 25 && bmiVal < 30) bmiLabel = 'Overweight';
  else if (bmiVal >= 30) bmiLabel = 'Obese';
 
  return { bmi: bmiVal, bmiLabel, bmr, tdee, goalCal, protG, carbG, fatG };
}
 
function fitnessLevel(u) {
  // Simple heuristic: BMI + age → beginner/intermediate/advanced suggestion
  const s = calcStats(u);
  if (s.bmi > 30 || u.age > 55) return 'beginner';
  if (s.bmi < 18.5 || u.age < 18) return 'beginner';
  if (u.activity >= 1.55) return 'intermediate';
  return 'beginner';
}

let selectedGoal = 'lose';
function selGoal(el) {
  document.querySelectorAll('.ob-goal').forEach(g => g.classList.remove('sel'));
  el.classList.add('sel');
  selectedGoal = el.dataset.goal;
}
 
function submitOnboarding() {
  const name   = document.getElementById('ob-name').value.trim();
  const age    = parseInt(document.getElementById('ob-age').value);
  const height = parseInt(document.getElementById('ob-height').value);
  const weight = parseInt(document.getElementById('ob-weight').value);
  const gender = document.getElementById('ob-gender').value;
  const activity = parseFloat(document.getElementById('ob-activity').value);
 
  let ok = true;
  function chk(id, errId, cond) {
    const el = document.getElementById(id);
    const er = document.getElementById(errId);
    if (cond) { el.classList.add('ob-err'); er.classList.add('show'); ok = false; }
    else { el.classList.remove('ob-err'); er.classList.remove('show'); }
  }
  chk('ob-name',   'e-name',   !name);
  chk('ob-age',    'e-age',    !age    || age<10    || age>90);
  chk('ob-height', 'e-height', !height || height<100 || height>250);
  chk('ob-weight', 'e-weight', !weight || weight<20  || weight>300);
  if (!ok) return;
 
  user = { name, age, height, weight, gender, activity, goal: selectedGoal };
  localStorage.setItem('fitlife-user', JSON.stringify(user));
  document.getElementById('onboarding').style.display = 'none';
  initApp();
}

function selectWorkout(el, key) {
  document.querySelectorAll('.w-card').forEach(c => c.classList.remove('selected'));
  if (el) el.classList.add('selected');
  curWorkout = workoutPlans.find(p => p.key === key);
  if (!curWorkout) return;
  clearInterval(timerInterval); timerRunning = false;
  timerLeft = timerTotal = curWorkout.time;
  document.getElementById('timer-start-btn').innerHTML = '<i class="ti ti-player-play"></i> Start';
  updateTimerDisplay();
  renderExercises();
}
 
function renderExercises() {
  if (!curWorkout) return;
  document.getElementById('exercise-list').innerHTML = curWorkout.exercises.map((e,i) => `
    <div class="ex-row">
      <div class="ex-num">${i+1}</div>

      <div class="ex-status" id="ex-s-${i}" onclick="this.classList.toggle('done')" role="checkbox" aria-label="Mark done"></div>

function toggleTimer() {
  if (timerRunning) {
    clearInterval(timerInterval); timerRunning = false;
    document.getElementById('timer-start-btn').innerHTML = '<i class="ti ti-player-play"></i> Resume';
  } else {
    if (timerLeft <= 0) { resetTimer(); return; }
    timerRunning = true;
    document.getElementById('timer-start-btn').innerHTML = '<i class="ti ti-player-pause"></i> Pause';
    timerInterval = setInterval(() => {
      timerLeft--;
      updateTimerDisplay();
      if (timerLeft <= 0) {
        clearInterval(timerInterval); timerRunning = false;
        document.getElementById('timer-start-btn').innerHTML = '<i class="ti ti-check"></i> Done!';
      }
    }, 1000);
  }
}
function resetTimer() {
  clearInterval(timerInterval); timerRunning = false;
  timerLeft = timerTotal = curWorkout ? curWorkout.time : 0;
  document.getElementById('timer-start-btn').innerHTML = '<i class="ti ti-player-play"></i> Start';
  updateTimerDisplay();
}
function updateTimerDisplay() {
  const m = Math.floor(timerLeft/60), s = timerLeft%60;
  document.getElementById('timer-display').textContent = String(m).padStart(2,'0')+':'+String(s).padStart(2,'0');
  document.getElementById('timer-progress').style.width = (timerLeft/Math.max(timerTotal,1)*100)+'%';
}

function renderFoods() {
  const el = document.getElementById('food-list');
  if (foodList.length === 0) {
    el.innerHTML = '<div style="font-size:13px;color:var(--text-secondary);padding:8px 0">No meals logged yet. Add your first meal below.</div>';
  } else {
    el.innerHTML = foodList.map((f,i) => `
      <div class="food-item">
        <div class="food-dot" style="background:${f.color}"></div>
        <div class="food-name">${f.n}</div>
        <div class="food-cal">${f.c} kcal</div>
        <button class="food-del" onclick="removeFood(${i})" aria-label="Remove"><i class="ti ti-x"></i></button>
      </div>`).join('');
  }
  document.getElementById('meal-count').textContent = foodList.length+' items';
}
 
function removeFood(i) { foodList.splice(i,1); renderFoods(); updateMacros(); }
 
function updateMacros() {
  const total = foodList.reduce((a,f) => a+f.c, 0);
  const pct = v => Math.min(Math.round(v*100),100);
  document.getElementById('cal-val').textContent  = total;
  document.getElementById('cal-bar').style.width  = pct(total/dietGoals.cal)+'%';
  const pG = foodList.filter(f=>f.t==='protein').reduce((a,f)=>a+Math.round(f.c/4),0);
  const cG = foodList.filter(f=>f.t==='carb').reduce((a,f)=>a+Math.round(f.c/4),0);
  const fG = foodList.filter(f=>f.t==='fat').reduce((a,f)=>a+Math.round(f.c/9),0);
  document.getElementById('prot-val').textContent = pG+'g';
  document.getElementById('carb-val').textContent = cG+'g';
  document.getElementById('fat-val').textContent  = fG+'g';
  document.getElementById('prot-bar').style.width = pct(pG/dietGoals.prot)+'%';
  document.getElementById('carb-bar').style.width = pct(cG/dietGoals.carb)+'%';
  document.getElementById('fat-bar').style.width  = pct(fG/dietGoals.fat)+'%';
}
 
function addFood() {
  const n = document.getElementById('food-name-input').value.trim();
  const c = parseInt(document.getElementById('food-cal-input').value);
  const t = document.getElementById('food-macro').value;
  const err = document.getElementById('food-err');
  if (!n||!c||c<1) { err.style.display='block'; return; }
  err.style.display='none';
  foodList.push({n,c,t,color:macroColors[t]});
  document.getElementById('food-name-input').value='';
  document.getElementById('food-cal-input').value='';
  renderFoods(); updateMacros();
}

function submitContact() {

  const msg   = document.getElementById('c-message').value.trim();

  if(!msg){se('err-message','c-message',true);ok=false;}else se('err-message','c-message',false);

    document.querySelector('.contact-wrap').querySelectorAll('input,textarea,select,button').forEach(e=>e.disabled=true);
    document.getElementById('contact-success').classList.add('show');
  }
}
